<script>

// ================= INIT AFTER PAGE LOAD =================
document.addEventListener("DOMContentLoaded", function () {

    // ================= ADD FILTER =================
    document.getElementById("addFilter").onclick = function () {

        let container = document.getElementById("filterContainer");

        let row = document.createElement("div");
        row.classList.add("d-flex", "mb-2");

        row.innerHTML = `
            <select class="form-control mr-2 key">
                <option value="Source_Type">Source_Type</option>
                <option value="Sso_Code">Sso_Code</option>
                <option value="Facility">Facility</option>
                <option value="Origin_Type">Origin_Type</option>
                <option value="Origin_Facility">Origin_Facility</option>
                <option value="Location">Location</option>
                <option value="Status">Status</option>
                <option value="Responsible_Person">Responsible_Person</option>
                <option value="Nuclide">Nuclide</option>
                <option value="Source_Model">Source_Model</option>
                <option value="Device_Model">Device_Model</option>
                <option value="Container_Type">Container_Type</option>
                <option value="Parent">Parent</option>
                <option value="Child">Child</option>
                <option value="Comment">Comment</option>
            </select>

            <input type="text" class="form-control mr-2 value" placeholder="Value">

            <button class="btn btn-danger remove">X</button>
        `;

        container.appendChild(row);

        row.querySelector(".remove").onclick = () => row.remove();
    };


    // ================= APPLY FILTERS =================
    document.getElementById("applyFilters").onclick = function () {

        let keys = document.querySelectorAll(".key");
        let values = document.querySelectorAll(".value");

        let params = new URLSearchParams();

        keys.forEach((k, i) => {
            if(values[i].value){
                params.append("key", k.value);
                params.append("value", values[i].value);
            }
        });

        window.location.href = "?" + params.toString();
    };


    // ================= SEARCH =================
    document.getElementById("searchInput").addEventListener("keyup", function () {
        let value = this.value.toLowerCase();

        document.querySelectorAll("#tableBody tr").forEach(row => {
            row.style.display =
                row.innerText.toLowerCase().includes(value) ? "" : "none";
        });
    });


    // ================= PAGE SIZE =================
    document.getElementById("pageSize").onchange = function () {
    let params = new URLSearchParams(window.location.search);
    params.set("size", this.value);
    params.delete("page"); // reset page when size changes
    window.location.search = params.toString();
};


    // ================= CSV =================
    window.exportCSV = function () {
        let params = window.location.search;
        window.location.href = "/export-csv/" + params;
    };


    // ================= SHOW / HIDE =================
   // ================= CHECKBOX SHOW/HIDE =================
document.querySelectorAll(".col-toggle").forEach(cb => {

    cb.addEventListener("change", function () {

        let col = this.dataset.col;
        let visible = this.checked;

        document.querySelectorAll(`[data-col="${col}"]`).forEach(el => {
            el.style.display = visible ? "" : "none";
        });

        // save to DB
        fetch("/save-column/", {
            method: "POST",
            headers: {
                "X-CSRFToken": "{{ csrf_token }}",
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                key: col,
                value: visible
            })
        });

    });

});
    // ================= LOAD SAVED COLUMNS =================
    fetch("/get-column/")
    .then(res => res.json())
    .then(data => {
        data.forEach(item => {

            document.querySelectorAll(`[data-col="${item.key}"]`).forEach(el => {
                el.style.display = item.value ? "" : "none";
            });

        });
    });

});
</script>
